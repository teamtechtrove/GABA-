// ── STREAMING RESPONSE ──────────────────────────────────────────────
async function streamResponse(system, messages, modelShort) {
  const response = await fetch('https://api.anthropic.com/v1/messages', {
    method: 'POST',
    headers: {
      'x-api-key': cfg.apiKey,
      'anthropic-version': '2023-06-01',
      'anthropic-dangerous-direct-browser-access': 'true',
      'content-type': 'application/json',
    },
    body: JSON.stringify({
      model: cfg.model,
      max_tokens: 4096,
      stream: true,
      system,
      messages,
    }),
  });

  if (!response.ok) {
    const err = await response.json().catch(() => ({ error: { message: `HTTP ${response.status}` } }));
    throw new Error(err?.error?.message || `HTTP ${response.status}`);
  }

  // Create bot bubble for streaming
  const { row, bubble } = createMsgBubble('', 'bot', modelShort);
  document.getElementById('messages').appendChild(row);

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let fullText = '';
  let buffer = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split('\n');
    buffer = lines.pop() || '';

    for (const line of lines) {
      if (!line.startsWith('data: ')) continue;
      const data = line.slice(6).trim();
      if (data === '[DONE]') continue;
      try {
        const parsed = JSON.parse(data);
        if (parsed.type === 'content_block_delta' && parsed.delta?.type === 'text_delta') {
          fullText += parsed.delta.text;
          bubble.innerHTML = formatMsg(fullText);
          scrollChat();
        }
      } catch(_) {}
    }
  }

  history.push({ role: 'assistant', content: fullText });
  addCopyButtons(bubble);
}

      'x-api-key': cfg.apiKey,
      'anthropic-version': '2023-06-01',
      'anthropic-dangerous-direct-browser-access': 'true',
      'content-type': 'application/json',
    },
    body: JSON.stringify({

      max_tokens: 4096,
      system,
      messages,
    }),
  });

  if (!response.ok) {

  }

  const data = await response.json();
  const text = data.content?.[0]?.text || '(empty response)';
  appendMsg(text, 'bot', modelShort);
  history.push({ role: 'assistant', content: text });
}

function createMsgBubble(text, sender, badge) {
  const row = document.createElement('div');
  row.className = `msg-row ${sender}`;

  const av = document.createElement('div');
  av.className = 'msg-av';
  av.textContent = sender === 'user' ? '👤' : '⚡';

  const body = document.createElement('div');

  const meta = document.createElement('div');
  meta.className = 'msg-meta';
  const t = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
  meta.innerHTML = `<span>${t}</span>`;
  if (badge) {
    const cls = badge === 'SAFETY' ? 'safe-badge' : badge === 'ERROR' ? 'err-badge' : '';
    meta.innerHTML += `<span class="prov-badge ${cls}">${esc(badge)}</span>`;
  }

  const bubble = document.createElement('div');
  bubble.className = 'msg-bubble';
  bubble.innerHTML = text ? formatMsg(text) : '';

  body.appendChild(meta);

  row.appendChild(av);
  row.appendChild(body);
  return { row, bubble };
}

function renderAllMessages() {
  const msgs = document.getElementById('messages');
  msgs.innerHTML = '';
  for (const turn of history) {
    if (turn.role === 'user') appendMsg(turn.content, 'user');
    else if (turn.role === 'assistant') appendMsg(turn.content, 'bot');
  }
}

function addCopyButtons(bubble) {
  if (!bubble) return;
  bubble.querySelectorAll('pre').forEach(pre => {
    if (pre.querySelector('.copy-btn')) return;
    const wrap = document.createElement('div');
    wrap.className = 'code-wrapper';
    pre.parentNode.insertBefore(wrap, pre);
    wrap.appendChild(pre);
    const btn = document.createElement('button');
    btn.className = 'copy-btn';
    btn.textContent = 'Copy';
    btn.onclick = () => {
      const code = pre.querySelector('code')?.textContent || pre.textContent;
      navigator.clipboard.writeText(code).then(() => { btn.textContent = 'Copied!'; setTimeout(() => btn.textContent = 'Copy', 2000); });
    };
    wrap.appendChild(btn);
  });
}

// ── MARKDOWN FORMATTER ──────────────────────────────────────────────
function formatMsg(t) {
  // Code blocks first
  t = t.replace(/```(\w*)\n?([\s\S]*?)```/g, (_, lang, code) =>
    `<pre><code class="lang-${lang || 'text'}">${esc(code.trim())}</code></pre>`

);
// Inline code
t = t.replace(/`([^`\n]+)`/g, (_, c) => `<code>${esc(c)}</code>`); // Headers t = t.replace(/^#{3}\s+(.+)$/gm, '<h3>$1</h3>'); t = t.replace(/^#{2}\s+(.+)$/gm, '<h2>$1</h2>'); t = t.replace(/^#{1}\s+(.+)$/gm, '<h1>$1</h1>'); // Bold & italic t = t.replace(/\*\*\*(.+?)\*\*\*/g, '<strong><em>$1</em></strong>'); t = t.replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>'); t = t.replace(/\*(.+?)\*/g, '<em>$1</em>'); // Blockquote t = t.replace(/^>\s+(.+)$/gm, '<blockquote>$1</blockquote>'); // Tables (basic) t = t.replace(/^\|(.+)\|$/gm, row => { if (/^[\|\s\-:]+$/.test(row)) return ''; const cells = row.split('|').filter((_, i, a) => i > 0 && i < a.length - 1); return '<tr>' + cells.map(c => `<td>${c.trim()}</td>`).join(’’) + ‘</tr>’;
});
t = t.replace(/((<tr>.*?</tr>\n?)+)/gs, ‘<table>$1</table>’);
// Lists
t = t.replace(/^[*-]\s+(.+)$/gm, ‘<li>$1</li>’);
t = t.replace(/(<li>[\s\S]*?</li>(?:\n<li>[\s\S]*?</li>)*)/g, ‘<ul>$1</ul>’);
t = t.replace(/^\d+.\s+(.+)$/gm, ‘<li>$1</li>’);
// URLs
t = t.replace(/(https?://[^\s<>”]+)/g, ‘<a href="$1" target="_blank" rel="noopener">$1</a>’);
// Newlines (outside block elements)
t = t.replace(/\n/g, ‘<br>’);
// Clean up excess breaks around block elements
t = t.replace(/<br>(<(?:pre|ul|ol|table|blockquote|h[1-6]))/g, ‘$1’);
t = t.replace(/((?:pre|ul|ol|table|blockquote|h[1-6])>)<br>/g, ‘$1’);
return t;
}

function esc(s) {
return String(s).replace(/&/g,’&’).replace(/</g,’<’).replace(/>/g,’>’).replace(/”/g,’"’);
}

// ── ANIMATED BACKGROUND CANVAS ──────────────────────────────────────
function initCanvas() {
const canvas = document.getElementById(‘bgCanvas’);
const ctx = canvas.getContext(‘2d’);
const N = 60;
const nodes = [];

function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
resize();
window.addEventListener(‘resize’, resize);

for (let i = 0; i < N; i++) {
nodes.push({
x: Math.random() * canvas.width,
y: Math.random() * canvas.height,
vx: (Math.random() - .5) * .45,
vy: (Math.random() - .5) * .45,
r: Math.random() * 2.2 + .8,
});
}

function draw() {
ctx.clearRect(0, 0, canvas.width, canvas.height);
nodes.forEach(n => {
n.x += n.vx; n.y += n.vy;
if (n.x < 0 || n.x > canvas.width)  n.vx *= -1;
if (n.y < 0 || n.y > canvas.height) n.vy *= -1;
});
for (let i = 0; i < N; i++) {
for (let j = i + 1; j < N; j++) {
const dx = nodes[i].x - nodes[j].x;
const dy = nodes[i].y - nodes[j].y;
const d = Math.sqrt(dx*dx + dy*dy);
if (d < 140) {
ctx.beginPath();
ctx.strokeStyle = `rgba(108,71,255,${.18 * (1 - d/140)})`;
ctx.lineWidth = .7;
ctx.moveTo(nodes[i].x, nodes[i].y);
ctx.lineTo(nodes[j].x, nodes[j].y);
ctx.stroke();
}
}
}
nodes.forEach(n => {
ctx.beginPath();
ctx.arc(n.x, n.y, n.r, 0, Math.PI*2);
ctx.fillStyle = ‘rgba(139,111,255,.5)’;
ctx.fill();
});
requestAnimationFrame(draw);
}
draw();
}
